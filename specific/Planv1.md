│ function MathOperationBar({ latex, onResult }) {                                                                                                                                                                   │
│   const ce = useComputeEngine();                                                                                                                                                                                   │
│   const expr = ce.parse(latex);                                                                                                                                                                                    │

│   const operations = [                                                                                                                                                                                             │

│   ];                                                                                                                                                                                                               │
│                                                                                                                                                                                                                    │
│   return (                                                                                                                                                                                                         │
│     <div className="operation-bar">                                                                                                                                                                                │
│       {operations.map(op => (                                                                                                                                                                                      │
│         <button key={op.label} onClick={() => {                                                                                                                                                                    │
│           const result = op.fn();                                                                                                                                                                                  │
│           onResult(result.latex);                                                                                                                                                                                  │
│         }}>                                                                                                                                                                                                        │
│           {op.label}                                                                                                                                                                                               │
│         </button>                                                                                                                                                                                                  │
│       ))}                                                                                                                                                                                                          │
│     </div>                                                                                                                                                                                                         │
│   );                                                                                                                                                                                                               │
